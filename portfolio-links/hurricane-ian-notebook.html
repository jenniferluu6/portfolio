def read_rgb_tif(path, bands=(1, 2, 3)):
    """
    Read selected bands from a GeoTIFF and return as H x W x C array.
    bands uses 1-based indexing as in rasterio.
    """
    with rasterio.open(path) as src:
        arr = src.read(list(bands))  # shape: (bands, rows, cols)
    return np.transpose(arr, (1, 2, 0))  # -> (rows, cols, bands)


def percentile_stretch(img, lower=2, upper=98):
    """
    Simple percentile stretch for visualization.
    Works band-by-band and returns float image in [0, 1].
    """
    img = img.astype(np.float32)
    out = np.zeros_like(img, dtype=np.float32)

    for i in range(img.shape[2]):
        band = img[:, :, i]
        lo = np.percentile(band, lower)
        hi = np.percentile(band, upper)

        if hi <= lo:
            out[:, :, i] = 0
        else:
            out[:, :, i] = np.clip((band - lo) / (hi - lo), 0, 1)

    return out

def make_square_bounds(geom, buffer_pct=0.20):
    """
    Turn one building polygon into its minimum enclosing square bbox,
    then expand it by buffer_pct on each side.
    """
    minx, miny, maxx, maxy = geom.bounds
    width = maxx - minx
    height = maxy - miny
    side_raw = max(width, height)
    side_raw = max(side_raw, 1e-6)  # guard against degenerate polygons

    cx = (minx + maxx) / 2.0
    cy = (miny + maxy) / 2.0

    side_buffered = side_raw * (1 + 2 * buffer_pct)
    half = side_buffered / 2.0

    square_bounds = (cx - half, cy - half, cx + half, cy + half)
    return square_bounds, side_raw, side_buffered


def crop_resize_rgb(src, bounds, target_size=(128, 128)):
    """
    Crop an RGB patch from map bounds and resample it directly to 128 x 128.
    Output shape is (H, W, 3).
    """
    left, bottom, right, top = bounds
    window = from_bounds(left, bottom, right, top, transform=src.transform)

    chip = src.read(
        indexes=[1, 2, 3],
        window=window,
        out_shape=(3, target_size[0], target_size[1]),
        resampling=Resampling.bilinear,
        boundless=True,
        fill_value=0,
    )
    return np.moveaxis(chip, 0, -1)


def build_bitemporal_dataset(
    label_path,
    pre_path,
    post_path,
    label_col="label",
    id_col="train_id",
    ignore_label=99,
    buffer_pct=0.20,
    target_size=(128, 128),
):
    labels_gdf = gpd.read_file(label_path)
    labels_gdf = labels_gdf[labels_gdf[label_col] != ignore_label].copy()
    labels_gdf[label_col] = labels_gdf[label_col].astype(int)

    with rasterio.open(pre_path) as pre_src, rasterio.open(post_path) as post_src:
        if pre_src.crs != post_src.crs:
            raise ValueError("Pre- and post-event rasters must use the same CRS.")

        labels_gdf = labels_gdf.to_crs(post_src.crs)

        metadata_records = []
        pre_chip_list = []
        post_chip_list = []

        for sample_idx, (_, row) in enumerate(labels_gdf.iterrows()):
            square_bounds, side_raw, side_buffered = make_square_bounds(
                row.geometry,
                buffer_pct=buffer_pct,
            )

            pre_chip = crop_resize_rgb(pre_src, square_bounds, target_size=target_size)
            post_chip = crop_resize_rgb(post_src, square_bounds, target_size=target_size)

            pre_chip_list.append(pre_chip)
            post_chip_list.append(post_chip)

            metadata_records.append({
                "sample_id": sample_idx,
                "train_id": row[id_col],
                "label": int(row[label_col]),
                "xmin": square_bounds[0],
                "ymin": square_bounds[1],
                "xmax": square_bounds[2],
                "ymax": square_bounds[3],
                "square_side_raw": side_raw,
                "square_side_buffered": side_buffered,
            })

    metadata_df = pd.DataFrame(metadata_records)
    pre_chips = np.asarray(pre_chip_list, dtype=np.uint8)
    post_chips = np.asarray(post_chip_list, dtype=np.uint8)
    y = metadata_df["label"].to_numpy(dtype=np.int32)

    # Keep metadata readable and tensors aligned by row index for later splitting/training.
    dataset = {
        "metadata": metadata_df,
        "pre": pre_chips,
        "post": post_chips,
        "y": y,
    }
    return dataset

def build_baseline_model_for_loading(input_shape=INPUT_SHAPE):
    """Rebuild the original xBD baseline in a form that matches the saved weights."""
    inputs = tf.keras.Input(shape=input_shape, name="input_1")

    # Branch 1: shallow CNN for local texture and roof-level details
    branch_1 = layers.Conv2D(32, (5, 5), activation="relu", padding="same", name="conv2d_1")(inputs)
    branch_1 = layers.MaxPooling2D((2, 2), name="max_pooling2d_2")(branch_1)
    branch_1 = layers.Conv2D(64, (3, 3), activation="relu", padding="same", name="conv2d_2")(branch_1)
    branch_1 = layers.MaxPooling2D((2, 2), name="max_pooling2d_3")(branch_1)
    branch_1 = layers.Conv2D(64, (3, 3), activation="relu", padding="same", name="conv2d_3")(branch_1)
    branch_1 = layers.MaxPooling2D((2, 2), name="max_pooling2d_4")(branch_1)
    branch_1 = layers.Flatten(name="flatten_1")(branch_1)

    # Branch 2: ResNet50 backbone for higher-level semantic/context features
    resnet_backbone = tf.keras.applications.ResNet50(
        include_top=False,
        weights=None,
        input_tensor=inputs,
    )
    branch_2 = layers.Flatten(name="flatten_2")(resnet_backbone.output)

    merged = layers.Concatenate(name="concatenate_1")([branch_1, branch_2])
    x = layers.Dense(2024, activation="relu", name="dense_1")(merged)
    x = layers.Dense(524, activation="relu", name="dense_2")(x)
    x = layers.Dense(124, activation="relu", name="dense_3")(x)
    outputs = layers.Dense(4, activation="softmax", name="dense_4")(x)

    return Model(inputs=inputs, outputs=outputs, name="xbd_baseline_classifier")

3. Fine-tuning¶

3.1 Model 3: Fine-tuning model, with only post-iamge¶

For this first fine-tuning experiment, we use the xBD baseline as a pretrained starting point and load the saved weights from classification.hdf5. We keep the ResNet50 backbone frozen so that its pretrained high-level feature extractor remains fixed. The trainable part includes the shallow CNN branch (conv2d_1-conv2d_3), the original classification head (dense_1-dense_3), and a new 3-class output layer that replaces the original 4-class output. In other words, this strategy performs post-event, transfer-learning-based fine-tuning by adapting only the lighter task-specific layers to the local label scheme while preserving the heavier pretrained backbone.

3.2 Model 4: bi-temporal model (fine-tuning), with pre/post images¶

For the bi-temporal model, both pre-event and post-event images are passed through a shared encoder initialized from the pretrained xBD baseline. The encoder reuses the original feature extractor up to concatenate_1. Within this encoder, the ResNet50 branch is frozen, while the shallow CNN branch remains trainable. The two encoded feature vectors are fused as concat([f_pre, f_post, abs(f_post - f_pre)]), and a new task-specific dense head is trained to predict the local 3-class damage labels. This strategy preserves the pretrained backbone while allowing the lighter layers and the new fusion head to adapt to local bi-temporal damage patterns.

# Encoder = original feature extractor up to concatenate_1
shared_encoder = Model(
    inputs=bt_base.input,
    outputs=bt_base.get_layer("concatenate_1").output,
    name="xbd_shared_encoder",
)

# Freeze the ResNet50 branch inside the encoder, but keep the shallow CNN branch trainable
encoder_non_resnet_layers = {
    "input_1",
    "conv2d_1",
    "max_pooling2d_2",
    "conv2d_2",
    "max_pooling2d_3",
    "conv2d_3",
    "max_pooling2d_4",
    "flatten_1",
    "flatten_2",
    "concatenate_1",
}

for layer in shared_encoder.layers:
    if layer.name not in encoder_non_resnet_layers:
        layer.trainable = False

for layer_name in ["conv2d_1", "conv2d_2", "conv2d_3"]:
    shared_encoder.get_layer(layer_name).trainable = True

# Two inputs share the same encoder weights
pre_input = Input(shape=(128, 128, 3), name="pre_input")
post_input = Input(shape=(128, 128, 3), name="post_input")

f_pre = shared_encoder(pre_input)
f_post = shared_encoder(post_input)

# Fusion: concatenate pre features, post features, and absolute feature differences
f_abs_diff = Lambda(lambda tensors: tf.abs(tensors[0] - tensors[1]), name="abs_feature_diff")([f_post, f_pre])
fused_features = Concatenate(name="fusion_concat")([f_pre, f_post, f_abs_diff])

# New task-specific head for local 3-class prediction
x = Dense(512, activation="relu", name="fusion_dense_1")(fused_features)
x = Dense(128, activation="relu", name="fusion_dense_2")(x)
bt_outputs = Dense(3, activation="softmax", name="bt_predictions")(x)

def build_full_area_bitemporal_dataset(
    building_path,
    pre_path,
    post_path,
    buffer_pct=0.20,
    target_size=(128, 128),
):
    """Create pre/post chips for every building polygon in the full study area."""
    buildings_gdf = gpd.read_file(building_path).reset_index(drop=True)
    original_building_count = len(buildings_gdf)

    candidate_id_cols = ["osm_id", "osmid", "id", "fid", "FID", "OBJECTID"]
    polygon_id_col = next((col for col in candidate_id_cols if col in buildings_gdf.columns), None)

    with rasterio.open(pre_path) as pre_src, rasterio.open(post_path) as post_src:
        if pre_src.crs != post_src.crs:
            raise ValueError("Pre- and post-event rasters must use the same CRS.")

        buildings_gdf = buildings_gdf.to_crs(post_src.crs)

        # Keep only building polygons that overlap the post-image extent, and clip them to that extent.
        raster_extent_geom = box(*post_src.bounds)
        buildings_gdf = buildings_gdf[buildings_gdf.geometry.intersects(raster_extent_geom)].copy()
        buildings_gdf["geometry"] = buildings_gdf.geometry.intersection(raster_extent_geom)
        buildings_gdf = buildings_gdf[~buildings_gdf.geometry.is_empty].copy()
        buildings_gdf = buildings_gdf.reset_index(drop=True)

        metadata_records = []
        pre_chip_list = []
        post_chip_list = []

        for sample_idx, row in buildings_gdf.iterrows():
            square_bounds, side_raw, side_buffered = make_square_bounds(
                row.geometry,
                buffer_pct=buffer_pct,
            )

            pre_chip = crop_resize_rgb(pre_src, square_bounds, target_size=target_size)
            post_chip = crop_resize_rgb(post_src, square_bounds, target_size=target_size)

            pre_chip_list.append(pre_chip)
            post_chip_list.append(post_chip)

            polygon_id = row[polygon_id_col] if polygon_id_col is not None else sample_idx
            metadata_records.append({
                "sample_id": sample_idx,
                "polygon_id": polygon_id,
                "xmin": square_bounds[0],
                "ymin": square_bounds[1],
                "xmax": square_bounds[2],
                "ymax": square_bounds[3],
                "square_side_raw": side_raw,
                "square_side_buffered": side_buffered,

    metadata_df = pd.DataFrame(metadata_records)
    pre_chips = np.asarray(pre_chip_list, dtype=np.uint8)
    post_chips = np.asarray(post_chip_list, dtype=np.uint8)
    return buildings_gdf, metadata_df, pre_chips, post_chips, original_building_count

Discussion and Policy Implications¶

The bi-temporal model achieved the best overall performance in this study, although its improvement over the post-only fine-tuned model was relatively modest. This limited gap is likely related in part to the small size of the held-out test set (69). Nevertheless, the broader result is consistent and important: the two fine-tuned models (Models 3 and 4) performed substantially better than the two non-adapted baselines, demonstrating the effectiveness of the proposed approach. These findings suggest that a pretrained damage assessment model developed from data in other regions can be efficiently adapted to a local disaster context through transfer learning with a relatively small amount of manually labeled local data. From a practical perspective, this provides a realistic and scalable pipeline for smaller disaster-affected cities that may not have the resources to build a large training dataset from scratch, but still require rapid building-level damage assessment.

From a modeling perspective, the bi-temporal model remains especially promising for broader deployment because it is likely to be more robust in full-area prediction for its dual input. It also showed strong ability to identify damaged buildings: the diagnostic results indicate high recall for damage-related labels, and when the task is simplified to a binary distinction between no damage and damage, the model achieves high accuracy. This creates an additional practical pathway for operational use. In real post-disaster settings, the model could be used not only for multi-class damage categorization, but also as a rapid binary damage detector to identify affected buildings and support prioritization in emergency response and recovery planning.
